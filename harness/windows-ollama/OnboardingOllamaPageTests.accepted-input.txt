    [Theory]
    [InlineData(false)]
    [InlineData(true)]
    public async Task MountedCloseOrQuitRetainsInstallerLoggerAndSetupLockUntilSettlement(bool quit)
    {
        await WithWindowAsync(async context =>
        {
            context.Host.HoldInstallation = true;
            await OpenOllamaAsync(context);
            var dialog = await DialogAsync(context.Page);
            var consent = Named<CheckBox>(dialog, "OllamaAppInstallConsent");
            Assert.IsAssignableFrom<IToggleProvider>(FrameworkElementAutomationPeer.CreatePeerForElement(consent)
                .GetPattern(PatternInterface.Toggle)).Toggle();
            Invoke(Button(dialog, "PrimaryButton"));
            await context.Host.InstallStarted.Task.WaitAsync(FixtureDeadline);
            var exited = false;
            Task cleanup;
            WindowManager? manager = null;
            if (quit)
            {
                manager = CreateManager();
                typeof(WindowManager).GetField("_setupWindow", BindingFlags.Instance | BindingFlags.NonPublic)!
                    .SetValue(manager, context.Window);
                // Use the actual shutdown coordinator, window manager, mounted page and setup lock.
                // Suppress only final Application.Exit so the suite's shared WinUI process survives.
                cleanup = new AppShutdownCoordinator().ShutdownAsync(new AppShutdownPlan(
                    manager.BeginShutdown,
                    [new("window manager", async () => await manager.CloseForShutdownAsync())],
                    () => exited = true));
            }
            else
            {
                context.Window.Close();
                cleanup = context.Window.CleanupCompleted;
            }
            try
            {
                await context.Host.CancellationSeen.Task.WaitAsync(FixtureDeadline);
                Assert.False(cleanup.IsCompleted);
                Assert.False(exited);
                Assert.False(SetupRunLock.TryAcquire(context.DataDirectory, out var refused, out _));
                Assert.Null(refused);
                Assert.Throws<IOException>(() =>
                {
                    using var probe = File.Open(context.LogPath, FileMode.Open, FileAccess.ReadWrite, FileShare.None);
                });
                Assert.Equal(0, context.Completed());
            }
            finally { context.Host.FinishInstallation(); }
            await cleanup.WaitAsync(FixtureDeadline);
            if (manager is not null) await manager.CloseForShutdownAsync().WaitAsync(FixtureDeadline);
            Assert.Equal(quit, exited);
            Assert.True(SetupRunLock.TryAcquire(context.DataDirectory, out var released, out _));
            released!.Dispose();
            using (File.Open(context.LogPath, FileMode.Open, FileAccess.ReadWrite, FileShare.None)) { }
            Assert.Contains("CancelledInstallationPresent", await File.ReadAllTextAsync(context.LogPath));
            Assert.Equal(1, context.Host.Installs);
            Assert.Equal(0, context.Host.Opens);
            Assert.DoesNotContain("openclaw.setup.verify", context.Transport.Calls);
            Assert.Equal(0, context.Completed());
        });
    }

    private async Task WithWindowAsync(Func<Context, Task> assertion, bool includeMetadataProbe = false)
    {
        OnboardingNativeProof.AssertIsolatedRoots();
        using var temp = new TempDirectory("ollama-mounted-");
        var configPath = temp.Combine("setup.json");
        await File.WriteAllTextAsync(configPath, JsonSerializer.Serialize(new SetupConfig()));
        await ui.RunOnUIAsync(async () =>
        {
            var resources = OnboardingWindowsFlowTests.LoadProgressResources(
                Environment.GetEnvironmentVariable("OPENCLAW_REPO_ROOT")!);
            Application.Current.Resources.MergedDictionaries.Add(resources);
            SetupWindow? window = null;
            Host? host = null;
            try
            {
                var data = temp.Combine("data");
                var local = temp.Combine("local");
                window = OnboardingNativeProof.CreateWindow(() => new SetupWindow(configPath: configPath,
                    dataDir: data, localDataDir: local, commandLineArgs: [], startupRegistrationAllowed: false));
                var root = Assert.IsType<Grid>(window.Content);
                var frame = Assert.IsType<Frame>(root.FindName("RootFrame"));
                var transport = new Transport { IncludeMetadataProbe = includeMetadataProbe };
                host = new Host(temp.Path);
                var completed = 0;
                var logPath = temp.Combine("ollama.jsonl");
                var args = new AiSetupPageArgs(window.AccessDraft.Config, data, local,
                    () => false, () => throw new InvalidOperationException("App readiness is not AI completion."),
                    TransportFactory: () => transport,
                    CompleteVerifiedSetup: proof =>
                    {
                        Assert.Equal(Transport.Model, proof.ModelRef);
                        completed++;
                        return Task.CompletedTask;
                    })
                {
                    OllamaSetupFactory = client =>
                        (new OllamaWindowsSetup(host, client, window.AccessDraft), new SetupLogger(logPath))
                };
                frame.Navigate(typeof(AiSetupPage), args);
                var page = Assert.IsType<AiSetupPage>(frame.Content);
                OnboardingNativeProof.ActivateOwned(window);
                await TestSupport.WaitForRenderedConditionAsync(() =>
                    page.IsLoaded && page.XamlRoot is not null &&
                    TestSupport.FindDescendants<SetupActionCard>(Assert.IsType<ItemsControl>(page.FindName("PrepareChoices")))
                        .Any(card => card.IsEnabled), "mounted Ollama setup entrypoint");
                await assertion(new(window, page, host, transport, data, logPath, () => completed));
            }
            finally
            {
                host?.FinishInstallation();
                if (window is not null)
                {
                    if (!window.IsClosed) window.Close();
                    await window.CleanupCompleted.WaitAsync(FixtureDeadline);
                }
                Application.Current.Resources.MergedDictionaries.Remove(resources);
            }
        });
    }

    private async Task OpenOllamaAsync(Context context)
    {
        var choices = Assert.IsType<ItemsControl>(context.Page.FindName("PrepareChoices"));
        var card = Assert.Single(TestSupport.FindDescendants<SetupActionCard>(choices),
            item => TestSupport.FindDescendants<TextBlock>(item).Any(text => text.Text == "Ollama"));
        Assert.Equal("Set up", Assert.Single(TestSupport.FindDescendants<TextBlock>(card),
            text => text.Text == "Set up").Text);
        Assert.IsAssignableFrom<IInvokeProvider>(FrameworkElementAutomationPeer.CreatePeerForElement(card)
            .GetPattern(PatternInterface.Invoke)).Invoke();
        await DialogAsync(context.Page);
    }

    private static async Task<ContentDialog> DialogAsync(AiSetupPage page)
    {
        ContentDialog? dialog = null;
        await TestSupport.WaitForRenderedConditionAsync(() =>
        {
            dialog = typeof(AiSetupPage).GetField("_ollamaDialog", BindingFlags.NonPublic | BindingFlags.Instance)!
                .GetValue(page) as ContentDialog;
            return dialog is { IsLoaded: true } &&
                TestSupport.FindDescendants<Button>(dialog).Any(button => button.Name == "PrimaryButton") &&
                Named<TextBlock>(dialog, "OllamaAppStatus").Text != "Checking the selected Gateway and this PC…";
        }, "native Ollama review dialog");
        return dialog!;
    }

    private static T Named<T>(DependencyObject root, string name) where T : FrameworkElement =>
        Assert.Single(TestSupport.FindDescendants<T>(root), element => element.Name == name);
    private static Button Button(ContentDialog dialog, string name) => Named<Button>(dialog, name);
    private static void Invoke(Button button)
    {
        Assert.True(button.IsEnabled);
        Assert.IsAssignableFrom<IInvokeProvider>(new ButtonAutomationPeer(button).GetPattern(PatternInterface.Invoke)).Invoke();
    }

    private sealed record Context(SetupWindow Window, AiSetupPage Page, Host Host, Transport Transport,
        string DataDirectory, string LogPath, Func<int> Completed);
    private static OllamaWindowsObservation Missing => new(OllamaWindowsInstallationState.NoSupportedRegistration,
        OllamaWindowsActivity.NotActive, null, OllamaWindowsApiState.NotListening, 0);
    private static OllamaWindowsObservation Verified => new(OllamaWindowsInstallationState.Verified,
        OllamaWindowsActivity.NotActive, new("synthetic-install", "synthetic-version"), OllamaWindowsApiState.NotListening, 0);

    private sealed class Host(string directory) : IOllamaWindowsAppHost
    {
        public OllamaWindowsObservation Current { get; set; } = Missing;
        public int Observations { get; private set; }
        public int Downloads { get; private set; }
        public int Installs { get; private set; }
        public int Opens { get; private set; }
        public bool HoldInstallation { get; set; }
        public TaskCompletionSource InstallStarted { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
        public TaskCompletionSource CancellationSeen { get; } = new(TaskCreationOptions.RunContinuationsAsynchronously);
        private readonly TaskCompletionSource _finish = new(TaskCreationOptions.RunContinuationsAsynchronously);
        public void FinishInstallation() => _finish.TrySetResult();
        public Task<OllamaWindowsObservation> ObserveAsync(CancellationToken ct)
        {
            ct.ThrowIfCancellationRequested();
            Observations++;
            return Task.FromResult(Current);
        }
        public Task<OllamaWindowsInstaller> AcquireAsync(CancellationToken ct, Action beforeDownload)
        {
            ct.ThrowIfCancellationRequested();
            beforeDownload();
            Downloads++;
            var scratch = Path.Combine(directory, "synthetic-download");
            Directory.CreateDirectory(scratch);
            var file = Path.Combine(scratch, "fixture.bin");
            File.WriteAllText(file, "Synthetic fixture bytes. Never executed.");
            return Task.FromResult(new OllamaWindowsInstaller(scratch,
                new FileStream(file, FileMode.Open, FileAccess.Read, FileShare.Read)));
        }
        public async Task<CommandResult> InstallAsync(OllamaWindowsInstaller installer, CancellationToken ct, Action beforeAdmission)
        {
            ct.ThrowIfCancellationRequested();
            beforeAdmission();
            Installs++;
            using var registration = ct.Register(() => CancellationSeen.TrySetResult());
            InstallStarted.TrySetResult();
            if (HoldInstallation) await _finish.Task;
            Current = Verified with { Activity = OllamaWindowsActivity.Active, ApiState = OllamaWindowsApiState.Reachable };
            return new(0, "", "", TimeSpan.Zero, false);
        }
        public Task OpenAsync(OllamaWindowsInstallation installation, CancellationToken ct, Action beforeOpen)
        {
            ct.ThrowIfCancellationRequested();
            beforeOpen();
            Opens++;
            Current = Verified with { Activity = OllamaWindowsActivity.Active, ApiState = OllamaWindowsApiState.Reachable };
            return Task.CompletedTask;
        }
    }

    private sealed class Transport : IGatewayAiSetupTransport
    {
        public const string Model = "ollama/synthetic:local";
        public const string MetadataAction = "Metadata provider action";
        public bool IncludeMetadataProbe { get; init; }
        public GatewayAiSetupRoute Route { get; } = new("ui-proof", "agent", "authority", new string('A', 64),
            new string('B', 64), "agent:agent:main");
        public long Generation => 1;
        public bool IsConnected => true;
        public IReadOnlyCollection<string> OperatorScopes => ["operator.admin"];
        public IReadOnlyCollection<string> Methods => ["openclaw.setup.detect", "openclaw.setup.verify", "openclaw.setup.activate.start",
            "openclaw.setup.prepare.start", "wizard.next", "wizard.cancel"];
        public List<string> Calls { get; } = [];
        public bool HasModel { get; set; }
        public string? ActivatedModel { get; private set; }
        public string? ActivatedKind { get; private set; }
        private string? _session;
        private bool _showModelConfirmation;
        public int ModelDownloadAnswers { get; private set; }
        public Task<JsonElement> RequestAsync(string method, object parameters, int timeoutMs, CancellationToken ct)
        {
            ct.ThrowIfCancellationRequested();
            Calls.Add(method);
            var args = JsonSerializer.SerializeToElement(parameters);
            object result;
            switch (method)
            {
                case "openclaw.setup.detect":
                    result = new
                    {
                        candidates = HasModel ? new object[]
                        {
                            new { kind = "existing-model", label = "Current model", detail = "Configured", modelRef = Model, recommended = false },
                            new { kind = "provider-auto:ollama", label = "Ollama", detail = "Discovered", modelRef = Model, recommended = false }
                        } : [],
                        manualProviders = Array.Empty<object>(), prepareOptions = new object[]
                        {
                            new { id = "ollama", label = "Ollama", kind = "custom", brandId = "ollama", actionLabel = "Gateway-supplied action" },
                        }.Concat(IncludeMetadataProbe ? new object[]
                        {
                            new { id = "metadata-probe", label = "Metadata probe", kind = "custom", brandId = "codex", actionLabel = MetadataAction },
                        } : Array.Empty<object>()),
                        workspace = "/synthetic", configuredModel = HasModel ? Model : ActivatedModel,
                        setupComplete = HasModel || ActivatedModel is not null
                    };
                    break;
                case "openclaw.setup.activate.start":
                    ActivatedModel = args.GetProperty("modelRef").GetString();
                    ActivatedKind = args.GetProperty("kind").GetString();
                    Assert.Equal(Model, ActivatedModel);
                    result = new { sessionId = args.GetProperty("sessionId").GetString(), done = true, status = "done", modelActivation = new { modelRef = ActivatedModel } };
                    break;
                case "openclaw.setup.prepare.start":
                    Assert.Equal("ollama", args.GetProperty("authChoice").GetString());
                    _session = args.GetProperty("sessionId").GetString();
                    result = new { sessionId = _session, done = false, status = "running", step = new { id = "address", type = "text", title = "Existing Ollama connection", message = "Synthetic address prompt" } };
                    break;
                case "wizard.next":
                    if (args.TryGetProperty("answer", out _))
                    {
                        if (_showModelConfirmation) ModelDownloadAnswers++;
                        _showModelConfirmation = true;
                    }
                    result = _showModelConfirmation
                        ? new { sessionId = _session, done = false, status = "running", step = (object)new { id = "model-download", type = "confirm", title = "Download a model?", message = "Synthetic separate model download confirmation", initialValue = false } }
                        : new { sessionId = _session, done = false, status = "running", step = (object)new { id = "address", type = "text", title = "Existing Ollama connection", message = "Synthetic address prompt" } };
                    break;
                case "wizard.cancel": result = new { status = "cancelled" }; break;
                case "openclaw.setup.verify": result = new { ok = true, modelRef = ActivatedModel, latencyMs = 1.0 }; break;
                default: throw new InvalidOperationException("Unexpected fixture request: " + method);
            }
            return Task.FromResult(JsonSerializer.SerializeToElement(result));
        }
    }
