import { createEffect, createSignal } from "solid-js";

export function Good(props: { name: string }) {
  const [count, setCount] = createSignal(0);
  createEffect(count, (value) => {
    document.title = String(value);
  });
  return (
    <button onClick={() => setCount((value) => value + 1)}>
      {props.name} {count()}
    </button>
  );
}

export function GoodNamespaces(props: { handler: (event: Event) => void }) {
  return <div onFoo={(event) => props.handler(event)} class={["base", { active: true }]} />;
}
